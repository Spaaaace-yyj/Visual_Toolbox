cmake_minimum_required(VERSION 3.16)
project(VisualToolbox VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d aruco)
add_executable(visual-toolbox src/main.cpp src/board.cpp src/window.cpp src/screen_metrics.cpp src/pdf_export.cpp)
target_include_directories(visual-toolbox PRIVATE src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(visual-toolbox PRIVATE Qt${QT_VERSION_MAJOR}::Widgets ${OpenCV_LIBS})
target_compile_options(visual-toolbox PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(board-tests tests/board_tests.cpp src/board.cpp)
  target_include_directories(board-tests PRIVATE src ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(board-tests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets ${OpenCV_LIBS})
  add_test(NAME board-tests COMMAND board-tests)
  set_tests_properties(board-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(toolbox-tests tests/toolbox_tests.cpp src/board.cpp src/screen_metrics.cpp src/pdf_export.cpp)
  target_include_directories(toolbox-tests PRIVATE src ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(toolbox-tests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets ${OpenCV_LIBS})
  add_test(NAME toolbox-tests COMMAND toolbox-tests)
  set_tests_properties(toolbox-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS visual-toolbox RUNTIME DESTINATION bin)
